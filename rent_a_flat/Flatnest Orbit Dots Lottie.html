How to use

The same JSON powers every platform via its respective Lottie player.

Web · lottie-web<script src="https://cdnjs.cloudflare.com/ajax/libs/lottie-web/5.12.2/lottie.min.js"></script> <div id="loader" style="width:120px"></div> <script> lottie.loadAnimation({ container: document.getElementById('loader'), renderer: 'svg', loop: true, autoplay: true, path: 'orbit-dots.json' }); </script>
React · lottie-reactimport Lottie from 'lottie-react'; import orbit from './orbit-dots.json'; function Loader() { return <Lottie animationData={orbit} loop style={{ width: 120 }} />; }
React Native · lottie-react-nativeimport LottieView from 'lottie-react-native'; <LottieView source={require('./orbit-dots.json')} autoPlay loop style={{ width: 120, height: 120 }} />
iOS · lottie-ios (SwiftUI)import Lottie struct Loader: View { var body: some View { LottieView(animation: .named("orbit-dots")) .playing(loopMode: .loop) .frame(width: 120, height: 120) } }
Android · lottie-android (Compose)// Put orbit-dots.json in res/raw val composition by rememberLottieComposition( LottieCompositionSpec.RawRes(R.raw.orbit_dots) ) LottieAnimation( composition = composition, iterations = LottieConstants.IterateForever, modifier = Modifier.size(120.dp) )
